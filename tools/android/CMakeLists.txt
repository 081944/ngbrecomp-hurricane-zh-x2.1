cmake_minimum_required(VERSION 3.20)
project(ngb_android_probes C)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain for these probes")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_link_options("-Wl,-z,max-page-size=16384")
set(XBOXRECOMP_OPT_LEVEL -O2)
add_subdirectory(../../xboxrecomp/src/platform platform)
add_subdirectory(../../xboxrecomp/src/kernel kernel)
add_subdirectory(../../xboxrecomp/src/nv2a nv2a)
foreach(test veh_posix vcpu_posix thread_suspend_posix)
    add_executable(${test}_test ../../xboxrecomp/tests/${test}/test_main.c)
    target_link_libraries(${test}_test PRIVATE platform)
endforeach()
foreach(test kernel_memory_posix memory_layout_posix)
    add_executable(${test}_test ../../xboxrecomp/tests/${test}/test_main.c)
    target_link_libraries(${test}_test PRIVATE xbox_kernel)
endforeach()
target_sources(memory_layout_posix_test PRIVATE memory_video_stubs.c)
add_executable(vulkan_caps vulkan_caps.c)
target_link_libraries(vulkan_caps PRIVATE vulkan)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/recomp/gen/recomp_types.h")
    # Link only the real context-switch assembly from fibers.c. Guest-specific
    # entry points are discarded by section GC; no game code or data is linked.
    add_executable(fiber_context_test fiber_context_test.c ../../src/fibers.c)
    target_link_libraries(fiber_context_test PRIVATE m)
    file(GLOB generated CONFIGURE_DEPENDS "../../src/recomp/gen/*.c")
    add_library(game_code_compile OBJECT EXCLUDE_FROM_ALL ${generated})
    target_include_directories(game_code_compile PRIVATE ../../xboxrecomp/src
        ../../xboxrecomp/include ../../xboxrecomp/src/kernel)
    target_compile_options(game_code_compile PRIVATE -O2 -fno-strict-aliasing -fwrapv
        -Wno-unused-label -Wno-unused-variable -Wno-unused-but-set-variable
        -Wno-parentheses-equality -Wno-tautological-compare)
endif()
